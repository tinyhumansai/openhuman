# Code View Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A read-only Code panel beside the chat that browses the thread's working folder and reviews git changes (diff vs `HEAD`) made by agents.

**Architecture:** A new core domain `code_view` (Rust, `crates/openhuman-core/src/code_view/`) exposes 5 read-only JSON-RPC controllers; it resolves the root from the thread's working folder, guards every path, and shells out to `git` with fixed arguments. The frontend adds `app/src/features/codeView/` (API wrapper, diff parser, highlighter, panel components), a `codeView` Redux slice with a status thunk, and minimal wiring in `Conversations.tsx` and `AssistantUiToolCall.tsx`.

**Tech Stack:** Rust (tokio, serde, `git` CLI), React 19 + TypeScript, Redux Toolkit, Vitest + Testing Library, `lowlight` / `highlight.js` (already in the lockfile).

**Spec:** `docs/superpowers/specs/2026-10-21-code-view-design.md` (read section 9 first: it records the corrections found while planning).

```rust
//! Root resolution, path guard and plain file access for the code view.
//!
//! Every path a caller supplies is relative to the root (the thread's working folder). It is checked
//! lexically, joined, canonicalized, and must stay inside the root and outside the paths the security
//! policy never opens. This module only reads; nothing here writes.

use std::path::{Component, Path, PathBuf};

use crate::security::SecurityPolicy;

pub(crate) const NOT_ALLOWED: &str = "path not allowed";
pub(crate) const NOT_FOUND: &str = "path not found";
/// Largest file `read_file` returns; larger files are truncated.
pub(crate) const MAX_READ_BYTES: u64 = 1024 * 1024;
/// A NUL byte in this many leading bytes marks a file as binary.
pub(crate) const BINARY_SNIFF_BYTES: usize = 8192;

/// Canonical form of a usable root: an existing directory outside the protected locations.
pub(crate) fn validate_root(root: &Path) -> Result<PathBuf, String> {
    let canonical = root
        .canonicalize()
        .map_err(|err| format!("working folder is not accessible: {err}"))?;
    if !canonical.is_dir() {
        return Err("working folder is not a directory".to_string());
    }
    if SecurityPolicy::is_always_forbidden(&canonical) {
        return Err("working folder is in a protected location".to_string());
    }
    Ok(canonical)
}

/// Lexical check shared by every RPC taking `rel_path`: relative, no `..`, no NUL.
pub(crate) fn check_rel_path(rel_path: &str) -> Result<&Path, String> {
    let rel = Path::new(rel_path);
    let bad = rel_path.contains('\0')
        || rel.is_absolute()
        || rel
            .components()
            .any(|c| !matches!(c, Component::Normal(_) | Component::CurDir));
    if bad {
        Err(NOT_ALLOWED.to_string())
    } else {
        Ok(rel)
    }
}

/// Resolve `rel_path` below the canonical `root`, refusing symlinks that point outside it.
pub(crate) fn resolve_in_root(root: &Path, rel_path: &str) -> Result<PathBuf, String> {
    let rel = check_rel_path(rel_path)?;
    let canonical = root
        .join(rel)
        .canonicalize()
        .map_err(|_| NOT_FOUND.to_string())?;
    if !canonical.starts_with(root) || SecurityPolicy::is_always_forbidden(&canonical) {
        return Err(NOT_ALLOWED.to_string());
    }
    Ok(canonical)
}

pub(crate) fn is_binary(bytes: &[u8]) -> bool {
    bytes[..bytes.len().min(BINARY_SNIFF_BYTES)].contains(&0)
}

#[cfg(test)]
#[path = "ops_tests.rs"]
mod tests;
```

/// One directory level, directories first then case-insensitive name order; `.git` is hidden.
pub(crate) async fn list_dir(root: &Path, rel_path: &str) -> Result<DirListing, String> {
    let dir = resolve_in_root(root, rel_path)?;
    let mut reader = tokio::fs::read_dir(&dir)
        .await
        .map_err(|err| format!("cannot list folder: {err}"))?;
    let mut entries = Vec::new();
    while let Some(entry) = reader
        .next_entry()
        .await
        .map_err(|err| format!("cannot list folder: {err}"))?
    {
        let name = entry.file_name().to_string_lossy().into_owned();
        if name == ".git" {
            continue;
        }
        let file_type = entry
            .file_type()
            .await
            .map_err(|err| format!("cannot list folder: {err}"))?;
        let (kind, size) = if file_type.is_symlink() {
            (EntryKind::Symlink, None)
        } else if file_type.is_dir() {
            (EntryKind::Dir, None)
        } else {
            (EntryKind::File, entry.metadata().await.ok().map(|m| m.len()))
        };
        entries.push(DirEntry { name, kind, size });
    }
    entries.sort_by(|a, b| {
        (a.kind != EntryKind::Dir)
            .cmp(&(b.kind != EntryKind::Dir))
            .then_with(|| a.name.to_lowercase().cmp(&b.name.to_lowercase()))
    });
    Ok(DirListing { entries })
}

/// File contents up to [`MAX_READ_BYTES`]; binary files return no content.
pub(crate) async fn read_file(root: &Path, rel_path: &str) -> Result<FileContent, String> {
    let path = resolve_in_root(root, rel_path)?;
    let meta = tokio::fs::metadata(&path)
        .await
        .map_err(|err| format!("cannot read file: {err}"))?;
    if !meta.is_file() {
        return Err("not a file".to_string());
    }
    let size = meta.len();
    let file = tokio::fs::File::open(&path)
        .await
        .map_err(|err| format!("cannot read file: {err}"))?;
    let mut buf = Vec::with_capacity(size.min(MAX_READ_BYTES) as usize);
    file.take(MAX_READ_BYTES)
        .read_to_end(&mut buf)
        .await
        .map_err(|err| format!("cannot read file: {err}"))?;
    if is_binary(&buf) {
        return Ok(FileContent { content: None, size, truncated: false, binary: true });
    }
    Ok(FileContent {
        content: Some(String::from_utf8_lossy(&buf).into_owned()),
        size,
        truncated: size > MAX_READ_BYTES,
        binary: false,
    })
}

//! shells out to `git` with fixed arguments. Every invocation runs with `-C <root>`, literal pathspecs,
//! no fsmonitor hook, no optional index locks, a 10 second timeout, and kills the child on timeout.

use std::path::Path;
use std::process::Stdio;
use std::time::Duration;

use super::types::ChangeStatus;

pub(crate) const GIT_TIMEOUT: Duration = Duration::from_secs(10);
/// Largest diff returned to the UI; longer diffs are cut on a line boundary.
pub(crate) const MAX_DIFF_BYTES: usize = 2 * 1024 * 1024;
/// The well-known empty tree, the diff base of a repository with no commits yet.
pub(crate) const EMPTY_TREE: &str = "4b825dc642cb6eb9a060e54bf8d69288fbee4904";

#[derive(Debug)]
pub(crate) struct GitOutput {
    pub status: Option<i32>,
    pub stdout: Vec<u8>,
    pub stderr: String,
}

pub(crate) async fn run_git(root: &Path, args: &[&str]) -> Result<GitOutput, String> {
    let mut cmd = tokio::process::Command::new("git");
    cmd.arg("-C")
        .arg(root)
        .args(["-c", "core.fsmonitor=false", "-c", "core.quotepath=false"])
        .args(args)
        .env("GIT_LITERAL_PATHSPECS", "1")
        .env("GIT_OPTIONAL_LOCKS", "0")
        .env("GIT_TERMINAL_PROMPT", "0")
        .stdin(Stdio::null())
        .stdout(Stdio::piped())
        .stderr(Stdio::piped())
        .kill_on_drop(true);
    let child = cmd
        .spawn()
        .map_err(|err| format!("git is not available: {err}"))?;
    match tokio::time::timeout(GIT_TIMEOUT, child.wait_with_output()).await {
        Ok(Ok(out)) => Ok(GitOutput {
            status: out.status.code(),
            stdout: out.stdout,
            stderr: String::from_utf8_lossy(&out.stderr).into_owned(),
        }),
        Ok(Err(err)) => Err(format!("git failed: {err}")),
        Err(_) => Err(format!("git timed out after {}s", GIT_TIMEOUT.as_secs())),
    }
}

/// One `status --porcelain=v2 -z` record; `path` is relative to the repository top level.
#[derive(Debug, Clone, PartialEq, Eq)]
pub(crate) struct StatusRecord {
    pub path: String,
    pub status: ChangeStatus,
    pub old_path: Option<String>,
}

pub(crate) fn parse_porcelain_v2(raw: &[u8]) -> Vec<StatusRecord> {
    let text = String::from_utf8_lossy(raw);
    let mut fields = text.split('\0').filter(|f| !f.is_empty());
    let mut out = Vec::new();
    while let Some(record) = fields.next() {
        let (tag, rest) = record.split_once(' ').unwrap_or((record, ""));
        match tag {
            // 1 XY sub mH mI mW hH hI path
            "1" => {
                let cols: Vec<&str> = rest.splitn(8, ' ').collect();
                if let [xy, .., path] = cols.as_slice() {
                    out.push(StatusRecord {
                        path: (*path).to_string(),
                        status: ordinary_status(xy),
                        old_path: None,
                    });
                }
            }
            // 2 XY sub mH mI mW hH hI Xscore path, then origPath as the next NUL field
            "2" => {
                let cols: Vec<&str> = rest.splitn(9, ' ').collect();
                let old_path = fields.next().map(str::to_string);
                if let [.., score, path] = cols.as_slice() {
                    let status = if score.starts_with('C') {
                        ChangeStatus::Added
                    } else {
                        ChangeStatus::Renamed
                    };
                    out.push(StatusRecord { path: (*path).to_string(), status, old_path });
                }
            }
            // u XY sub m1 m2 m3 mW h1 h2 h3 path
            "u" => {
                if let Some(path) = rest.splitn(10, ' ').nth(9) {
                    out.push(StatusRecord {
                        path: path.to_string(),
                        status: ChangeStatus::Conflicted,
                        old_path: None,
                    });
                }
            }
            "?" => out.push(StatusRecord {
                path: rest.to_string(),
                status: ChangeStatus::Untracked,
                old_path: None,
            }),
            _ => {}
        }
    }
    out
}

fn ordinary_status(xy: &str) -> ChangeStatus {
    let mut chars = xy.chars();
    let (x, y) = (chars.next().unwrap_or('.'), chars.next().unwrap_or('.'));
    if x == 'A' {
        ChangeStatus::Added
    } else if x == 'D' || y == 'D' {
        ChangeStatus::Deleted
    } else {
        ChangeStatus::Modified
    }
}

/// `diff --numstat -z`: `added\tdeleted\tpath\0`, or `added\tdeleted\t\0old\0new\0` for a rename.
/// Binary files report `-`, mapped to `None`.
pub(crate) fn parse_numstat(raw: &[u8]) -> Vec<(String, Option<u64>, Option<u64>)> {
    let text = String::from_utf8_lossy(raw);
    let mut fields = text.split('\0');
    let mut out = Vec::new();
    while let Some(record) = fields.next() {
        if record.is_empty() {
            continue;
        }
        let mut cols = record.splitn(3, '\t');
        let (Some(added), Some(deleted), Some(path)) = (cols.next(), cols.next(), cols.next())
        else {
            continue;
        };
        let path = if path.is_empty() {
            let _old = fields.next();
            fields.next().unwrap_or_default().to_string()
        } else {
            path.to_string()
        };
        out.push((path, added.parse().ok(), deleted.parse().ok()));
    }
    out
}

pub(crate) fn count_lines(bytes: &[u8]) -> u64 {
    let newlines = bytes.iter().filter(|b| **b == b'\n').count() as u64;
    newlines + u64::from(!bytes.is_empty() && !bytes.ends_with(b"\n"))
}

pub(crate) fn truncate_diff(raw: Vec<u8>) -> (String, bool) {
    if raw.len() <= MAX_DIFF_BYTES {
        return (String::from_utf8_lossy(&raw).into_owned(), false);
    }
    let cut = raw[..MAX_DIFF_BYTES]
        .iter()
        .rposition(|b| *b == b'\n')
        .map_or(MAX_DIFF_BYTES, |i| i + 1);
    (String::from_utf8_lossy(&raw[..cut]).into_owned(), true)
}

    let out = result?;

/// Root path plus git branch/HEAD. A folder outside git (or no `git` binary) is `is_git: false`.
pub(crate) async fn root_info(root: &Path) -> RootInfo {
    let path = root.to_string_lossy().into_owned();
    let inside = stdout_line(run_git(root, &["rev-parse", "--is-inside-work-tree"]).await)
        .is_some_and(|line| line == "true");
    if !inside {
        return RootInfo { path, is_git: false, branch: None, head: None };
    }
    let branch = stdout_line(run_git(root, &["symbolic-ref", "--short", "-q", "HEAD"]).await);
    let head = stdout_line(run_git(root, &["rev-parse", "--short=7", "--verify", "-q", "HEAD"]).await);
    RootInfo { path, is_git: true, branch, head }
}

/// Files under the root that differ from `head` (or from the empty tree when unborn).
pub(crate) async fn status(root: &Path, head: Option<&str>) -> Result<Vec<ChangedFile>, String> {
    let prefix = stdout_line(run_git(root, &["rev-parse", "--show-prefix"]).await).unwrap_or_default();
    let records = parse_porcelain_v2(&ok_stdout(
        run_git(
            root,
            &[
                "-c",
                "status.relativePaths=false",
                "status",
                "--porcelain=v2",
                "-z",
                "--untracked-files=all",
                "--",
                ".",
            ],
        )
        .await,
        &[0],
    )?);
    let base = head.unwrap_or(EMPTY_TREE);
    let counts: HashMap<String, (Option<u64>, Option<u64>)> = parse_numstat(&ok_stdout(
        run_git(
            root,
            &[
                "diff",
                "--numstat",
                "-z",
                "-M",
                "--relative",
                "--no-ext-diff",
                "--no-textconv",
                base,
                "--",
                ".",
            ],
        )
        .await,
        &[0],
    )?)
    .into_iter()
    .map(|(path, added, deleted)| (path, (added, deleted)))
    .collect();

    let mut files = Vec::new();
    for record in records {
        let Some(path) = record.path.strip_prefix(prefix.as_str()).map(str::to_string) else {
            continue;
        };
        let old_path = record
            .old_path
            .and_then(|old| old.strip_prefix(prefix.as_str()).map(str::to_string));
        let (added, deleted) = if record.status == ChangeStatus::Untracked {
            (untracked_lines(root, &path).await, None)
        } else {
            counts.get(&path).copied().unwrap_or((None, None))
        };
        files.push(ChangedFile { path, status: record.status, added, deleted, old_path });
    }
    Ok(files)
}

async fn untracked_lines(root: &Path, rel_path: &str) -> Option<u64> {
    let path = ops::resolve_in_root(root, rel_path).ok()?;
    let meta = tokio::fs::metadata(&path).await.ok()?;
    if !meta.is_file() || meta.len() > ops::MAX_READ_BYTES {
        return None;
    }
    let bytes = tokio::fs::read(&path).await.ok()?;
    (!ops::is_binary(&bytes)).then(|| count_lines(&bytes))
}

/// Unified diff of one path against `head`; an untracked file diffs against `/dev/null`.
pub(crate) async fn diff(root: &Path, rel_path: &str, head: Option<&str>) -> Result<FileDiff, String> {
    let rel = ops::check_rel_path(rel_path)?;
    if SecurityPolicy::is_always_forbidden(&root.join(rel)) {
        return Err(ops::NOT_ALLOWED.to_string());
    }
    let tracked = run_git(root, &["ls-files", "--error-unmatch", "--", rel_path])
        .await
        .is_ok_and(|out| out.status == Some(0));
    // An untracked file diffs against /dev/null. `--no-index` reads the file system directly, so
    // only a file that resolves inside the root qualifies; anything else (a deletion staged with
    // `git rm`, a path that never existed) diffs against the base, which only reads git objects.
    let untracked = !tracked && ops::resolve_in_root(root, rel_path).is_ok();
    let raw = if !untracked {
        let base = head.unwrap_or(EMPTY_TREE);
        ok_stdout(
            run_git(
                root,
                &["diff", "--no-color", "--no-ext-diff", "--no-textconv", "-U3", "-M", base, "--", rel_path],
            )
            .await,
            &[0],
        )?
    } else {
        ok_stdout(
            run_git(
                root,
                &["diff", "--no-index", "--no-color", "--no-ext-diff", "--no-textconv", "-U3", "--", "/dev/null", rel_path],
            )
            .await,
            &[0, 1],
        )?
    };
    let (diff, truncated) = truncate_diff(raw);
    Ok(FileDiff { diff, truncated })
}

export type DiffLineKind = 'ctx' | 'add' | 'del' | 'meta';

export interface DiffLineEntry {
  kind: DiffLineKind;
  oldNo: number | null;
  newNo: number | null;
  text: string;
}

export interface DiffHunk {
  header: string;
  lines: DiffLineEntry[];
}

export interface ParsedDiff {
  hunks: DiffHunk[];
  binary: boolean;
}

const HUNK_HEADER = /^@@ -(\d+)(?:,\d+)? \+(\d+)(?:,\d+)? @@/;

/** Parse the unified diff of a single file (`git diff` output) into numbered hunks. */
export function parseUnifiedDiff(text: string): ParsedDiff {
  const hunks: DiffHunk[] = [];
  let binary = false;
  let current: DiffHunk | null = null;
  let oldNo = 0;
  let newNo = 0;
  for (const raw of text.split('\n')) {
    const header = HUNK_HEADER.exec(raw);
    if (header) {
      current = { header: raw, lines: [] };
      hunks.push(current);
      oldNo = Number(header[1]);
      newNo = Number(header[2]);
      continue;
    }
    if (!current) {
      if (raw.startsWith('Binary files ')) binary = true;
      continue;
    }
    if (raw.startsWith('+')) current.lines.push({ kind: 'add', oldNo: null, newNo: newNo++, text: raw.slice(1) });
    else if (raw.startsWith('-')) current.lines.push({ kind: 'del', oldNo: oldNo++, newNo: null, text: raw.slice(1) });
    else if (raw.startsWith(' ')) current.lines.push({ kind: 'ctx', oldNo: oldNo++, newNo: newNo++, text: raw.slice(1) });
    else if (raw.startsWith('\\')) current.lines.push({ kind: 'meta', oldNo: null, newNo: null, text: raw });
  }
  return { hunks, binary };
}

/** New-file line of the first change, used to open the editor at the right place. */
export function firstChangedLine(parsed: ParsedDiff): number | null {
  for (const hunk of parsed.hunks) {
    const added = hunk.lines.find(line => line.kind === 'add');
    if (added) return added.newNo;
    const first = hunk.lines.find(line => line.newNo !== null);
    if (first) return first.newNo;
  }
  return null;
}

export function isFileEditTool(name: string): boolean {
  return FILE_EDIT_TOOLS.has(name);
}

export function fileToolPath(args: unknown): string | null {
  if (!args || typeof args !== 'object') return null;
  const record = args as Record<string, unknown>;
  const value = record.path ?? record.file_path;
  return typeof value === 'string' && value.trim() !== '' ? value : null;
}

export function requestCodeView(path: string | null): void {
  window.dispatchEvent(new CustomEvent<CodeViewOpenDetail>(CODE_VIEW_OPEN_EVENT, { detail: { path } }));
}

/** Successful file-editing tool calls in a thread; a change in this count triggers a refresh. */
export function countCompletedFileEdits(
  entries: readonly { name: string; status: string }[] | undefined,
): number {
  return entries?.filter(entry => entry.status === 'success' && FILE_EDIT_TOOLS.has(entry.name)).length ?? 0;
}

/** `path` relative to `root` (relative inputs pass through), or null when it lies outside the root. */
export function toRelativePath(root: string, path: string): string | null {
  const normalize = (value: string) => value.replace(/\\/g, '/').replace(/\/+$/, '');
  const base = normalize(root);
  const target = normalize(path);
  if (!target.startsWith('/') && !/^[A-Za-z]:\//.test(target)) return target.replace(/^\.\//, '');
  if (target === base) return '';
  return target.startsWith(`${base}/`) ? target.slice(base.length + 1) : null;
}

```ts
import { createAsyncThunk, createSlice, type PayloadAction } from '@reduxjs/toolkit';

import { type ChangedFile, codeViewApi, type RootInfo } from '../features/codeView/codeViewApi';
import { openArtifact } from './chatRuntimeSlice';
import { closeCodePanel, openCodePanel } from './codeViewActions';
import { resetUserScopedState } from './resetActions';

export type CodeViewTab = 'changes' | 'files';

export interface CodeViewState {
  open: boolean;
  threadId: string | null;
  tab: CodeViewTab;
  /** As given by the opener (absolute or relative); `CodePanel` makes it root-relative. */
  selectedPath: string | null;
  root: RootInfo | null;
  files: ChangedFile[];
  loading: boolean;
  error: string | null;
  updatedAt: number | null;
  refreshNonce: number;
  pendingRequestId: string | null;
}

export const initialCodeViewState: CodeViewState = {
  open: false,
  threadId: null,
  tab: 'changes',
  selectedPath: null,
  root: null,
  files: [],
  loading: false,
  error: null,
  updatedAt: null,
  refreshNonce: 0,
  pendingRequestId: null,
};

/** Root info plus changed files for a thread; files are empty outside git. */
export const loadCodeStatus = createAsyncThunk('codeView/loadStatus', async (threadId: string) => {
  const root = await codeViewApi.root(threadId);
  const files = root.is_git ? await codeViewApi.gitStatus(threadId) : [];
  return { threadId, root, files, updatedAt: Date.now() };
});

const codeViewSlice = createSlice({
  name: 'codeView',
  initialState: initialCodeViewState,
  reducers: {
    syncCodeViewThread(state, action: PayloadAction<string | null>) {
      if (state.threadId === action.payload) return;
      return {
        ...initialCodeViewState,
        open: state.open && action.payload !== null,
        tab: state.tab,
        threadId: action.payload,
        refreshNonce: state.refreshNonce,
      };
    },
    setCodeViewTab(state, action: PayloadAction<CodeViewTab>) {
      state.tab = action.payload;
    },
    selectCodePath(state, action: PayloadAction<string | null>) {
      state.selectedPath = action.payload;
    },
    requestCodeRefresh(state) {
      state.refreshNonce += 1;
    },
  },
  extraReducers: builder => {
    builder
      .addCase(openCodePanel, (state, action) => {
        state.open = true;
        if (state.threadId !== action.payload.threadId) {
          state.threadId = action.payload.threadId;
          state.root = null;
          state.files = [];
          state.selectedPath = null;
        }
        if (action.payload.path) {
          state.selectedPath = action.payload.path;
          state.tab = 'changes';
        }
      })
      .addCase(closeCodePanel, state => {
        state.open = false;
      })
      .addCase(openArtifact, state => {
        state.open = false;
      })
      .addCase(loadCodeStatus.pending, (state, action) => {
        state.loading = true;
        state.pendingRequestId = action.meta.requestId;
      })
      .addCase(loadCodeStatus.fulfilled, (state, action) => {
        if (action.meta.requestId !== state.pendingRequestId || action.payload.threadId !== state.threadId) return;
        state.loading = false;
        state.pendingRequestId = null;
        state.error = null;
        state.root = action.payload.root;
        state.files = action.payload.files;
        state.updatedAt = action.payload.updatedAt;
      })
      .addCase(loadCodeStatus.rejected, (state, action) => {
        if (action.meta.requestId !== state.pendingRequestId) return;
        state.loading = false;
        state.pendingRequestId = null;
        state.error = action.error.message ?? 'error';
      })
      .addCase(resetUserScopedState, () => initialCodeViewState);
  },
});

export const { syncCodeViewThread, setCodeViewTab, selectCodePath, requestCodeRefresh } = codeViewSlice.actions;
export { closeCodePanel, openCodePanel };
export default codeViewSlice.reducer;
```

import { useMemo } from 'react';

import { Alert, AlertDescription } from '../../components/ui';
import { useT } from '../../lib/i18n/I18nContext';
import { highlightLine, languageForPath } from './highlight';
import { type DiffLineKind, parseUnifiedDiff } from './parseUnifiedDiff';

const ROW_CLASS: Record<DiffLineKind, string> = {
  ctx: '',
  add: 'bg-sage-500/15',
  del: 'bg-coral-500/15',
  meta: 'text-content-faint',
};
const SIGN: Record<DiffLineKind, string> = { ctx: '', add: '+', del: '-', meta: '' };
const SIGN_CLASS: Record<DiffLineKind, string> = {
  ctx: '',
  add: 'text-sage-600',
  del: 'text-coral-600',
  meta: '',
};

interface DiffViewProps {
  diff: string;
  path: string;
  truncated: boolean;
}

export function DiffView({ diff, path, truncated }: DiffViewProps) {
  const { t } = useT();
  const parsed = useMemo(() => parseUnifiedDiff(diff), [diff]);
  const language = languageForPath(path);

  if (parsed.binary) return <p className="m-4 text-sm text-content-muted">{t('codeView.diff.binary')}</p>;
  if (parsed.hunks.length === 0) return <p className="m-4 text-sm text-content-muted">{t('codeView.diff.empty')}</p>;

  return (
    <div className="font-mono text-xs leading-relaxed">
      {truncated && (
        <Alert variant="warning" density="compact" className="m-2">
          <AlertDescription>{t('codeView.diff.truncated')}</AlertDescription>
        </Alert>
      )}
      {parsed.hunks.map((hunk, hunkIndex) => (
        <div key={hunkIndex}>
          <div className="bg-primary-500/10 px-3 py-0.5 text-content-muted">{hunk.header}</div>
          {hunk.lines.map((line, lineIndex) => (
            <div key={lineIndex} data-kind={line.kind} className={`flex whitespace-pre ${ROW_CLASS[line.kind]}`}>
              <span className="w-10 shrink-0 select-none pr-2 text-right text-content-faint">{line.oldNo ?? ''}</span>
              <span className="w-10 shrink-0 select-none pr-2 text-right text-content-faint">{line.newNo ?? ''}</span>
              <span className={`w-5 shrink-0 select-none text-center ${SIGN_CLASS[line.kind]}`}>{SIGN[line.kind]}</span>
              <span className="min-w-0">{line.kind === 'meta' ? line.text : highlightLine(line.text, language)}</span>
            </div>
          ))}
        </div>
      ))}

  );

import { useMemo } from 'react';

import { Alert, AlertDescription } from '../../components/ui';

  /** Changes (e.g. the status `updatedAt`) reload every expanded folder. */
  refreshKey: number | null;
  onSelect: (path: string) => void;
}

type Loaded = DirEntry[] | 'error';

export function FileTree({ threadId, selectedPath, refreshKey, onSelect }: FileTreeProps) {
  const { t } = useT();
  const [children, setChildren] = useState<Record<string, Loaded>>({});
  const [expanded, setExpanded] = useState<Set<string>>(() => new Set(['']));

  const load = useCallback(
    async (dir: string) => {
      try {
        const entries = await codeViewApi.listDir(threadId, dir);
        setChildren(prev => ({ ...prev, [dir]: entries }));
      } catch {
        setChildren(prev => ({ ...prev, [dir]: 'error' }));
      }
    },
    [threadId],
  );

  useEffect(() => {
    setChildren({});
    setExpanded(new Set(['']));
    void load('');
  }, [load]);

  useEffect(() => {
    if (refreshKey === null) return;
    expanded.forEach(dir => void load(dir));
    // Reload on refresh only; `expanded` changes load folders on their own.
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [refreshKey]);

  const toggle = (dir: string) => {
    setExpanded(prev => {
      const next = new Set(prev);
      if (next.has(dir)) next.delete(dir);
      else {
        next.add(dir);
        if (!children[dir]) void load(dir);
      }
      return next;
    });
  };

  const renderDir = (dir: string, depth: number): ReactNode => {
    const entries = children[dir];
    if (entries === undefined) return null;
    if (entries === 'error') {
      return <p className="px-2 py-1 text-xs text-coral-600">{t('codeView.error.title')}</p>;
    }
    return entries.map(entry => {
      const path = dir ? `${dir}/${entry.name}` : entry.name;
      const isDir = entry.kind === 'dir';
      const isOpen = isDir && expanded.has(path);
      const Icon = isDir ? (isOpen ? ChevronDown : ChevronRight) : entry.kind === 'symlink' ? Link2 : File;
      return (
        <div key={path}>
          <button
            type="button"
            role="treeitem"
            aria-expanded={isDir ? isOpen : undefined}
            aria-selected={path === selectedPath}
            onClick={() => (isDir ? toggle(path) : onSelect(path))}
            style={{ paddingInlineStart: 8 + depth * 12 }}
            className={`flex w-full items-center gap-1.5 rounded-md py-1 pe-2 text-left text-xs hover:bg-surface-muted ${
              path === selectedPath ? 'bg-primary-500/15' : ''
            }`}>
            <Icon className="h-3.5 w-3.5 shrink-0 text-content-muted" />
            {isDir && <Folder className="h-3.5 w-3.5 shrink-0 text-content-muted" />}
            <span className="truncate">{entry.name}</span>
          </button>
          {isOpen && renderDir(path, depth + 1)}
        </div>
      );
    });
  };

  return (
    <div role="tree" aria-label={t('codeView.tree.aria')} className="p-1.5">
      {renderDir('', 0)}
    </div>
  );
}

import { type KeyboardEvent, type MouseEvent, useCallback, useEffect, useRef, useState } from 'react';

export const MIN_WIDTH = 360;
export const MAX_WIDTH = 1400;
export const DEFAULT_WIDTH = 760;
const STEP = 24;

const clamp = (width: number) => Math.min(MAX_WIDTH, Math.max(MIN_WIDTH, width));

/** Width of a right-edge panel, resized by dragging its left border or with the arrow keys. */
export function usePanelWidth() {
  const [width, setWidth] = useState(DEFAULT_WIDTH);
  const drag = useRef<{ startX: number; startWidth: number } | null>(null);

  useEffect(() => {
    const onMove = (event: globalThis.MouseEvent) => {
      if (drag.current) setWidth(clamp(drag.current.startWidth + drag.current.startX - event.clientX));
    };
    const onUp = () => {
      drag.current = null;
    };
    window.addEventListener('mousemove', onMove);
    window.addEventListener('mouseup', onUp);
    return () => {
      window.removeEventListener('mousemove', onMove);
      window.removeEventListener('mouseup', onUp);
    };
  }, []);

  const onMouseDown = useCallback(
    (event: MouseEvent) => {
      event.preventDefault();
      drag.current = { startX: event.clientX, startWidth: width };
    },
    [width],
  );

  const onKeyDown = useCallback((event: KeyboardEvent) => {
    if (event.key === 'ArrowLeft') setWidth(w => clamp(w + STEP));
    else if (event.key === 'ArrowRight') setWidth(w => clamp(w - STEP));
  }, []);

  return { width, onMouseDown, onKeyDown };
}

type Remote<T> =
  | { status: 'loading' }
  | { status: 'ready'; value: T }
  | { status: 'error'; message: string };

/** Load `fetcher()` whenever `key` changes; a response for an older key is dropped. */
function useRemote<T>(key: string, fetcher: () => Promise<T>): Remote<T> {
  const [state, setState] = useState<Remote<T>>({ status: 'loading' });
  useEffect(() => {
    let cancelled = false;
    setState({ status: 'loading' });
    fetcher().then(
      value => {
        if (!cancelled) setState({ status: 'ready', value });
      },
      (error: unknown) => {
        if (!cancelled) setState({ status: 'error', message: error instanceof Error ? error.message : String(error) });
      },
    );
    return () => {
      cancelled = true;
    };
    // `key` encodes every input of `fetcher`.
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [key]);
  return state;
}

export function vscodeUrl(root: string, rel: string, line: number | null): string {
  const absolute = `${root.replace(/\\/g, '/').replace(/\/+$/, '')}/${rel}`;
  const path = absolute.startsWith('/') ? absolute.slice(1) : absolute;
  return `vscode://file/${encodeURI(path)}${line ? `:${line}` : ''}`;

    if (!threadId) return;
    const timer = window.setTimeout(() => void dispatch(loadCodeStatus(threadId)), RELOAD_DELAY_MS);

  }, [dispatch, threadId, edits, refreshNonce]);

  return null;
}
